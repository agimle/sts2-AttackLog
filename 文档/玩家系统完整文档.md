### 1.1 平台类型

```csharp
namespace MegaCrit.Sts2.Core.Platform;

public enum PlatformType
```

### 1.2 平台工具类 PlatformUtil

`PlatformUtil` 是平台相关操作的核心静态类，提供玩家名称获取、好友列表、邀请等功能。

**源文件位置**: `src/Core/Platform/PlatformUtil.cs`

```csharp
public static class PlatformUtil
{
    // 获取当前主要平台
    public static PlatformType PrimaryPlatform { get; }
    
    // 获取玩家显示名称（如Steam用户名）
    public static string GetPlayerName(PlatformType platformType, ulong playerId);
    
    // 获取本地玩家ID
    public static ulong GetLocalPlayerId(PlatformType platformType);
    
    // 获取有开放大厅的好友列表
    public static Task<IEnumerable<ulong>> GetFriendsWithOpenLobbies(PlatformType platformType);
    
    // 打开邀请对话框
    public static void OpenInviteDialog(INetGameService netService);
    
    // 打开URL
    public static void OpenUrl(string url);
    
    // 设置Rich Presence（Steam状态）
    public static void SetRichPresence(string token, string? playerGroup, int? groupSize);
    public static void SetRichPresenceValue(string key, string? value);
    public static void ClearRichPresence();
}
```

**使用示例**：

```csharp
using MegaCrit.Sts2.Core.Platform;

// 获取玩家Steam用户名
ulong steamId = 76561198000000000UL;
string playerName = PlatformUtil.GetPlayerName(PlatformType.Steam, steamId);

// 获取本地玩家ID
ulong myId = PlatformUtil.GetLocalPlayerId(PlatformType.Steam);

// 获取有开放大厅的好友
var friends = await PlatformUtil.GetFriendsWithOpenLobbies(PlatformType.Steam);
foreach (ulong friendId in friends)
{
    Console.WriteLine($"好友 {friendId} 有开放的大厅");
}

// 打开Steam邀请对话框
PlatformUtil.OpenInviteDialog(netGameService);
```

### 1.3 大厅玩家结构 LobbyPlayer

`LobbyPlayer` 是大厅中玩家信息的结构体，用于游戏开始前的玩家状态管理。

**源文件位置**: `src/Core/Entities/Multiplayer/LobbyPlayer.cs`

```csharp
namespace MegaCrit.Sts2.Core.Entities.Multiplayer;

public struct LobbyPlayer : IPacketSerializable
{
    /// <summary>
    /// 玩家唯一网络ID（Steam ID）
    /// </summary>
    public ulong id;
    
    /// <summary>
    /// 槽位ID（玩家在游戏中的位置索引）
    /// </summary>
    public int slotId;

    /// 选择的角色模型
    /// </summary>
    public CharacterModel character;
    
    /// <summary>
    /// 解锁状态（已解锁的卡牌、遗物等）
    /// </summary>
    public SerializableUnlockState unlockState;

    /// 最大解锁的多难度等级
    /// </summary>
    public int maxMultiplayerAscensionUnlocked;

    /// 是否已准备就绪
    /// </summary>

    public void Deserialize(PacketReader reader);

        return $"Player {id}, {character.Id.Entry}";
```

## 二、玩家信息获取

```csharp
namespace MegaCrit.Sts2.Core.Runs;

public interface IPlayerCollection

    /// <summary>
    /// 获取所有玩家的只读列表

    /// 获取玩家在集合中的槽位索引
```

### 2.2 通过 IRunState 获取玩家

`IRunState` 是游戏运行状态的核心接口，继承自 `IPlayerCollection`。

**源文件位置**: `src/Core/Runs/IRunState.cs`

```csharp
public interface IRunState : ICardScope, IPlayerCollection
```

**使用示例**：

```csharp

{
    Console.WriteLine($"玩家: {player.NetId}, 角色: {player.Character.Title}");
}

// 根据NetId获取特定玩家
Player? player = runState.GetPlayer(76561198000000000UL);
if (player != null)
{
    // 处理玩家
}

// 获取玩家槽位索引
int slotIndex = runState.GetPlayerSlotIndex(player);

// 判断是否多人游戏
bool isMultiplayer = runState.Players.Count > 1;
```

### 2.3 通过 CombatState 获取玩家

在战斗中可以通过 `CombatState` 获取玩家信息。

**源文件位置**: `src/Core/Combat/CombatState.cs`

```csharp
public class CombatState
{
    public IReadOnlyList<Player> Players { get; }
    
    /// <summary>
    /// 根据玩家ID获取玩家
    /// </summary>
    public Player? GetPlayer(ulong playerId)
    {
        return Players.FirstOrDefault(p => p.NetId == playerId);
    }
    
    /// <summary>
    /// 遍历所有Hook监听者
    /// </summary>
    public IEnumerable<AbstractModel> IterateHookListeners();
}
```

```csharp
public static class LocalContext
{
    /// <summary>
    /// 从玩家集合获取本地玩家
    /// </summary>
    public static Player? GetMe(IPlayerCollection? playerCollection)
    {
        if (!NetId.HasValue || playerCollection == null)
            return null;
        return playerCollection.GetPlayer(NetId.Value)

    }
```

```csharp
public static class LocalContext
{
    /// <summary>
    /// 判断是否是本地玩家

    }
    
    /// <summary>
    /// 判断生物是否属于本地玩家
    /// </summary>
    public static bool IsMe(Creature? creature)
    {
        return IsMe(creature?.Player);
    }
    
    /// <summary>
    /// 判断玩家集合中是否包含本地玩家
    /// </summary>
    public static bool ContainsMe(IEnumerable<Player> players)
    {
        return players.Any(IsMe);
    }
    
    /// <summary>
    /// 判断生物集合中是否包含本地玩家的生物
    /// </summary>
    public static bool ContainsMe(IEnumerable<Creature> creatures)
    {
        return creatures.Any(IsMe);
    }
    
    /// <summary>
    /// 判断卡牌是否属于本地玩家
    /// </summary>
    public static bool IsMine(CardModel? card)
    {
        return card != null && card.IsMutable && IsMe(card.Owner);
    }
    
    /// <summary>
    /// 判断药水是否属于本地玩家
    /// </summary>
    public static bool IsMine(PotionModel? potion)
    {
        return potion != null && potion.IsMutable && IsMe(potion.Owner);
    }
}
```

### 7.4 多人游戏玩家处理

```csharp
using MegaCrit.Sts2.Core.Context;
using MegaCrit.Sts2.Core.Platform;
using MegaCrit.Sts2.Core.Runs;

public static class MultiplayerHelper
{
    public static void ProcessAllPlayers(IRunState runState)
    {
        // 获取本地玩家
        Player? localPlayer = LocalContext.GetMe(runState);
        
        // 获取其他玩家
        var otherPlayers = runState.Players
            .Where(p => !LocalContext.IsMe(p))
            .ToList();
        
        Console.WriteLine($"本地玩家: {localPlayer?.Character.Title ?? "无"}");
        Console.WriteLine($"其他玩家数量: {otherPlayers.Count}");
        
        foreach (Player player in otherPlayers)
        {
            string name = PlatformUtil.GetPlayerName(PlatformType.Steam, player.NetId);
            Console.WriteLine($"  - {name} ({player.Character.Title})");
        }
    }
    
    public static Dictionary<ulong, string> GetPlayerNames(IRunState runState)
    {
        var names = new Dictionary<ulong, string>();
        
        foreach (Player player in runState.Players)
        {
            string name = PlatformUtil.GetPlayerName(PlatformType.Steam, player.NetId);
            names[player.NetId] = name;
        }
        
        return names;
    }
}
```
